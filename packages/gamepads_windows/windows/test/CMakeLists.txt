cmake_minimum_required(VERSION 3.14)
project(gamepads_rumble_test LANGUAGES CXX)
add_compile_options(/utf-8)
enable_testing()
add_executable(rumble_test rumble_test.cpp)
target_compile_features(rumble_test PRIVATE cxx_std_17)
target_compile_options(rumble_test PRIVATE /W4 /WX)
add_test(NAME rumble_stop_paths COMMAND rumble_test)
add_library(release_fixture SHARED release_fixture.cpp)
target_compile_features(release_fixture PRIVATE cxx_std_17)
add_executable(gameinput_release_test gameinput_release_test.cpp)
add_dependencies(gameinput_release_test release_fixture)
add_test(NAME gameinput_release_lifetime COMMAND gameinput_release_test)
add_executable(gamepad_stop_test gamepad_stop_test.cpp)
target_compile_features(gamepad_stop_test PRIVATE cxx_std_17)
add_test(NAME gamepad_stop_lifetime COMMAND gamepad_stop_test)
set_tests_properties(rumble_stop_paths gameinput_release_lifetime gamepad_stop_lifetime PROPERTIES TIMEOUT 15)
